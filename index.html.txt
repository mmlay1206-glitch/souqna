    function navigateTo(viewId) {

    // REAL-TIME LISTENER FOR PUBLIC PRODUCTS
    db.collection('products').orderBy('createdAt', 'desc').onSnapshot(snapshot => {

      applyFilters();

      console.warn("Firestore live index falling back:", error);
      // Fallback query without sorting if index is building
      db.collection('products').get().then(snap => {
        publicProducts = snap.docs.map(doc => ({ id: doc.id, ...doc.data() }));

    // FILTER & SEARCH PRODUCTS FOR VISITORS
    function applyFilters() {
      const searchVal = document.getElementById('searchInput').value.trim().toLowerCase();
      const regionVal = document.getElementById('regionFilter').value;
      const categoryVal = document.getElementById('categoryFilter').value;

      const filtered = publicProducts.filter(p => {
        const matchesSearch = !searchVal || 
          (p.title && p.title.toLowerCase().includes(searchVal)) || 
          (p.shopName && p.shopName.toLowerCase().includes(searchVal));
        const matchesRegion = !regionVal || p.region === regionVal;
        const matchesCategory = !categoryVal || p.category === categoryVal;
        return matchesSearch && matchesRegion && matchesCategory;
      });

      renderProducts(filtered);
    }

    // REGISTER HANDLER
    async function handleRegister(e) {
      e.preventDefault();
      const btn = document.getElementById('btnRegisterSubmit');
      btn.disabled = true;
      btn.innerText = 'جاري تسجيل المتجر...';

      const email = document.getElementById('regEmail').value.trim();
      const password = document.getElementById('regPassword').value;
      const shopName = document.getElementById('regShopName').value.trim();
      const region = document.getElementById('regRegion').value;
      const phone = document.getElementById('regPhone').value.trim();

      try {
        const cred = await auth.createUserWithEmailAndPassword(email, password);
        await db.collection('shops').doc(cred.user.uid).set({
          ownerId: cred.user.uid,
          name: shopName,
          region: region,
          phone: phone,
          createdAt: firebase.firestore.FieldValue.serverTimestamp()
        });
        alert('تم إنشاء متجرك بنجاح في قاعدة البيانات السحابية!');
        navigateTo('dashboard');
      } catch (err) {
        alert('خطأ في التسجيل: ' + err.message);
      } finally {
        btn.disabled = false;
        btn.innerText = 'تأكيد وإنشاء المتجر';
      }
    }

    // VENDOR PRODUCTS LOADER
    function loadVendorProducts() {
      if (!currentUser) return;
      db.collection('products').where('ownerId', '==', currentUser.uid).onSnapshot(snap => {
        const vGrid = document.getElementById('vendorProductsGrid');
        if (snap.empty) {
          vGrid.innerHTML = `<p class="col-span-full text-center text-slate-400 py-6 text-xs">لم تقم بإضافة أي منتجات لمتجرك بعد.</p>`;
          return;
        }

        vGrid.innerHTML = snap.docs.map(doc => {
          const item = doc.data();
          return `
            <div class="border border-slate-200 rounded-xl p-3 bg-slate-50 flex items-center justify-between gap-2">
              <div>
                <h4 class="font-bold text-slate-900 text-xs">${item.title}</h4>
                <p class="text-xs text-emerald-700 font-extrabold">${Number(item.price).toLocaleString()} ل.س</p>
              </div>
              <button onclick="deleteProduct('${doc.id}')" class="text-red-600 hover:bg-red-100 p-1.5 rounded-lg text-xs font-bold transition">
                حذف 🗑️
              </button>
            </div>
          `;
        }).join('');
      });
    }

    // SAVE PRODUCT
    async function handleSaveProduct(e) {
      e.preventDefault();
      if (!currentShop) return alert('خطأ: يتعذر العثور على بيانات متجرك');

      const btn = document.getElementById('btnSaveProd');
      btn.disabled = true;
      btn.innerText = 'جاري النشر...';

      const title = document.getElementById('prodTitle').value.trim();
      const price = Number(document.getElementById('prodPrice').value);
      const category = document.getElementById('prodCategory').value;
      const image = document.getElementById('prodImage').value.trim();
      const desc = document.getElementById('prodDesc').value.trim();

      try {
        await db.collection('products').add({
          ownerId: currentUser.uid,
          shopId: currentShop.id,
          shopName: currentShop.name,
          region: currentShop.region,
          phone: currentShop.phone,
          title, price, category, image, desc,
          createdAt: firebase.firestore.FieldValue.serverTimestamp()
        });

        closeAddProductModal();
        document.getElementById('prodTitle').value = '';
        document.getElementById('prodPrice').value = '';
        document.getElementById('prodDesc').value = '';
        alert('تم نشر المنتج بنجاح وظهر لجميع زوار موقع سوقنا!');
      } catch (err) {
        alert('حدث خطأ أثناء حفظ المنتج: ' + err.message);
      } finally {
        btn.disabled = false;
        btn.innerText = 'نشر للمشترين';
      }
    }

    // DELETE PRODUCT
    async function deleteProduct(id) {
      if (confirm('هل أنت تأكد من إزالة هذا المنتج من متجرك؟')) {
        try {
          await db.collection('products').doc(id).delete();
        } catch (err) {
          alert('خطأ في الحذف: ' + err.message);
        }
      }
    }
